## **🚀 Quick Start**

### Local Development (5 minutes)

```bash
# 1. Clone repo
git clone https://github.com/yourusername/codeconclave-pro.git
cd codeconclave-pro

# 2. Start infrastructure
docker-compose up -d

# 3. Setup backend
cd apps/api
npm install
cp .env.example .env
# Edit .env with your API keys
npx prisma migrate dev
npm run dev &

# 4. Setup frontend
cd ../web
npm install
npm run dev &

# 5. Open browser
# http://localhost:3000 (frontend)
# http://localhost:3001/health (API)
```

## **PART 1: LOCAL DEVELOPMENT**

### Step 1.1 — Clone & Install
```bash
git clone https://github.com/yourusername/codeconclave-pro.git
cd codeconclave-pro

# Install backend dependencies
cd apps/api
npm install

# Install frontend dependencies
cd ../web
npm install
```

### Step 1.2 — Database Setup
```bash
# Start infrastructure (Docker required)
docker-compose up -d

# Wait for containers to be healthy
docker-compose ps

# Apply database migrations
cd apps/api
npx prisma migrate dev
npx prisma generate
```

### Step 1.3 — Environment Configuration
```bash
# Backend
cp apps/api/.env.example apps/api/.env

# Edit .env with your keys:
```

### Step 2.5 — CDN & Edge (Optional but Recommended)

```bash
# Use Cloudflare for CDN
# 1. Add DNS records for your domain
# 2. Enable caching for static assets
# 3. Use Cloudflare Workers for rate limiting

# Cloudflare Worker example (wrangler.toml):
[env.production]
routes = [
  { pattern = "api.example.com/*", zone_name = "example.com" }
]

# In Cloudflare Worker:
export default {
  async fetch(request) {
    const ip = request.headers.get('CF-Connecting-IP');
    const url = new URL(request.url);
    
    // Rate limit: 100 req/min per IP
    const key = `ratelimit:${ip}`;
    const count = await RATE_LIMIT.get(key);
    if (count > 100) return new Response('Too many requests', { status: 429 });
    
    // Pass through to origin
    return fetch(request);
  }
};
```

```typescript
// apps/api/src/services/ai-router.ts

async function routeTask(task: {
  title: string;
  description: string;
  context: any;
}): Promise<{
  coworker: string;
  response: string;
  tokens: number;
  confidence: number;
}> {
  // 1. Analyze task intent
  const intent = await analyzeIntent(task.description);
  
  // 2. Select coworker
  const coworker = selectCoworker(intent);
  
  // 3. Get system prompt
  const systemPrompt = getSystemPrompt(coworker);
  
  // 4. Call primary model
  const primaryModel = getModel(coworker);
  let response;
  try {
    response = await primaryModel.invoke(systemPrompt, task);
  } catch (err) {
    // Fallback
    const fallbackModel = getFallbackModel(coworker);
    response = await fallbackModel.invoke(systemPrompt, task);
  }
  
  // 5. Log metrics
  logCoworkerMetrics(coworker, response.tokens, response.latency);
  
  // 6. Cache for future queries
  cacheResponse(task.description, response, intent);
  
  return response;
}

// Coworker selection logic
function selectCoworker(intent: string): string {
  if (intent.includes("architecture") || intent.includes("design")) return "ARCHITECT";
  if (intent.includes("code") || intent.includes("implement")) return "CODER";
  if (intent.includes("test") || intent.includes("qa")) return "TESTER";
  if (intent.includes("security") || intent.includes("threat")) return "SECURITY";
  if (intent.includes("performance") || intent.includes("optimize")) return "PERFORMANCE";
  return "ARCHITECT"; // Default
}
```

  const [issues, setIssues] = useState<DebuggingIssue[]>([]);

  const analyzeCode = async () => {
    setAnalyzing(true);
    try {
      const res = await api.post('/ai/analyze-code', { code });
      setIssues(res.data.issues || []);
      toast.success(`Found ${res.data.issues?.length || 0} issues`);
    } catch (err) {
      toast.error('Analysis failed');
    } finally {
      setAnalyzing(false);
    }
  };

  const applySuggestion = async (issueId: string) => {
    try {
      const issue = issues.find(i => i.id === issueId);
      if (issue?.fix) {
        setCode(prev => prev.replace(issue.location, issue.fix!));
        toast.success('Fix applied');
      }
    } catch (err) {
      toast.error('Failed to apply fix');
    }
  };

  const severityColor = (severity: string) => {
    switch (severity) {
      case 'critical': return 'text-red-500 bg-red-500/10';
      case 'warning': return 'text-orange-500 bg-orange-500/10';
      default: return 'text-blue-500 bg-blue-500/10';
    }
  };

  return (
    <div className="p-6 space-y-6">
      <motion.div initial={{ opacity: 0, y: -10 }} animate={{ opacity: 1, y: 0 }}>
        <h1 className="text-3xl font-black gold-gradient-text">AI Code Debugger</h1>
        <p className="text-conclave-gray-4 text-sm mt-1">Realtime intelligent issue detection</p>
      </motion.div>

      <div className="grid grid-cols-3 gap-6">
        {/* Code Input */}
        <div className="col-span-2 glass-panel rounded-xl p-4">
          <textarea
            value={code}
            onChange={(e) => setCode(e.target.value)}
            placeholder="Paste your code here..."
            className="w-full h-96 bg-conclave-darker border border-conclave-border rounded-lg p-4 text-conclave-gray-2 font-mono text-sm focus:border-conclave-gold focus:outline-none resize-none"
          />
          <motion.button
            whileHover={{ scale: 1.02 }}
            onClick={analyzeCode}
            disabled={analyzing || !code}
            className="w-full mt-4 py-2.5 rounded-lg bg-conclave-gold text-conclave-dark font-bold text-sm hover:bg-conclave-gold/90 disabled:opacity-50 flex items-center justify-center gap-2"
          >
            {analyzing ? (
              <>
                <Loader2 className="w-4 h-4 animate-spin" />
                Analyzing...
              </>
            ) : (
              <>
                <Brain className="w-4 h-4" />
                Analyze Code
              </>
            )}
          </motion.button>
        </div>

        {/* Issues List */}
        <div className="glass-panel rounded-xl p-4 h-full overflow-y-auto">
          <h2 className="text-sm font-bold text-white mb-4 flex items-center gap-2">
            <Bug className="w-4 h-4 text-conclave-gold" />
            Issues Found
          </h2>
          <div className="space-y-3">
            {issues.map((issue) => (
              <motion.div
                key={issue.id}
                initial={{ opacity: 0, x: 20 }}
                animate={{ opacity: 1, x: 0 }}
                className={`p-3 rounded-lg border border-conclave-border ${severityColor(issue.severity)}`}
              >
                <div className="flex items-start gap-2 mb-2">
                  <AlertTriangle className="w-4 h-4 mt-0.5 flex-shrink-0" />
                  <div className="flex-1">
                    <p className="text-xs font-semibold uppercase">{issue.type}</p>
                    <p className="text-xs mt-0.5">{issue.description}</p>
                  </div>
                </div>
                {issue.fix && (
                  <button
                    onClick={() => applySuggestion(issue.id)}
                    className="text-xs px-2 py-1 rounded bg-white/10 hover:bg-white/20 transition-colors"
                  >
                    Apply Fix
                  </button>
                )}
              </motion.div>
            ))}
            {issues.length === 0 && (
              <p className="text-xs text-conclave-gray-4 text-center py-8">No issues detected</p>
            )}
          </div>
        </div>
      </div>
    </div>
  );
}

function generateTokens(user: any) {
  const payload: JWTPayload = {
    userId: user.id,
    email: user.email,
    orgId: user.orgId,
    role: user.role,
    iat: Math.floor(Date.now() / 1000),
    exp: 0,
  };

  const accessToken = jwt.sign(
    { ...payload, exp: Math.floor(Date.now() / 1000) + 15 * 60 },
    process.env.JWT_SECRET || 'dev-secret-change-in-prod',
    { algorithm: 'HS256' }
  );

  const refreshToken = jwt.sign(
    { ...payload, exp: Math.floor(Date.now() / 1000) + 7 * 24 * 60 * 60 },
    process.env.JWT_REFRESH_SECRET || 'dev-refresh-secret',
    { algorithm: 'HS256' }
  );

  return { accessToken, refreshToken };
}

function hashPassword(password: string): Promise<string> {
  return bcrypt.hash(password, 12);
}

function verifyPassword(password: string, hash: string): Promise<boolean> {
  return bcrypt.compare(password, hash);
}

function encryptAPIKey(key: string): string {
  const iv = crypto.randomBytes(16);
  const cipher = crypto.createCipheriv('aes-256-gcm', Buffer.from(process.env.ENCRYPTION_KEY || 'dev-key-32-bytes-long!'), iv);
  let encrypted = cipher.update(key, 'utf8', 'hex');
  encrypted += cipher.final('hex');
  const authTag = cipher.getAuthTag();
  return iv.toString('hex') + ':' + authTag.toString('hex') + ':' + encrypted;
}

function decryptAPIKey(encrypted: string): string {
  const parts = encrypted.split(':');
  const iv = Buffer.from(parts[0], 'hex');
  const authTag = Buffer.from(parts[1], 'hex');
  const encryptedData = parts[2];
  const decipher = crypto.createDecipheriv('aes-256-gcm', Buffer.from(process.env.ENCRYPTION_KEY || 'dev-key-32-bytes-long!'), iv);
  decipher.setAuthTag(authTag);
  let decrypted = decipher.update(encryptedData, 'hex', 'utf8');
  decrypted += decipher.final('utf8');
  return decrypted;
}

function correlationMiddleware(req: Request, res: Response, next: NextFunction) {
  const correlationId = req.headers['x-correlation-id'] as string || `cc_${uuid()}`;
  (req as any).correlationId = correlationId;
  res.setHeader('X-Correlation-ID', correlationId);
  next();
}

async function authMiddleware(req: Request, res: Response, next: NextFunction) {
  try {
    const token = req.headers.authorization?.split(' ')[1];
    if (!token) {
      return res.status(401).json({ success: false, message: 'No token provided' });
    }
    const decoded = jwt.verify(token, process.env.JWT_SECRET || 'dev-secret-change-in-prod') as JWTPayload;
    (req as any).user = decoded;
    next();
  } catch (err) {
    return res.status(401).json({ success: false, message: 'Invalid token' });
  }
}

async function rateLimitMiddleware(req: Request, res: Response, next: NextFunction) {
  const key = `ratelimit:${(req as any).user?.userId || req.ip}`;
  const limit = 100; // 100 requests per minute
  const current = await redis.incr(key);
  if (current === 1) {
    await redis.expire(key, 60);
  }
  if (current > limit) {
    return res.status(429).json({ success: false, message: 'Rate limit exceeded' });
  }
  next();
}

app.post('/api/v1/auth/register', async (req: Request, res: Response) => {
  try {
    const parsed = RegisterSchema.parse(req.body);
    const existing = await prisma.user.findUnique({ where: { email: parsed.email } });
    if (existing) {
      return res.status(400).json({ success: false, message: 'Email already in use' });
    }

    const passwordHash = await hashPassword(parsed.password);
    const orgId = `org_${uuid()}`;
    const user = await prisma.user.create({
      data: {
        id: `user_${uuid()}`,
        email: parsed.email,
        name: parsed.name,
        passwordHash,
        orgId,
        role: 'USER',
      },
    });

    // Create default entitlement
    await prisma.entitlement.create({
      data: {
        userId: user.id,
        plan: 'free',
        tokensPerDay: 500,
        interactionsPerDay: 2,
        projectsLimit: 1,
        asyncHoursPerMonth: 0,
        expiresAt: new Date(Date.now() + 365 * 24 * 60 * 60 * 1000),
      },
    });

    const { accessToken, refreshToken } = generateTokens(user);
    await redis.setex(`refresh_${user.id}`, 7 * 24 * 60 * 60, refreshToken);

    logger.info('User registered', { userId: user.id, email: user.email, correlationId: (req as any).correlationId });
    return res.status(201).json({
      success: true,
      user: { id: user.id, email: user.email, name: user.name },
      accessToken,
      refreshToken,
    });
  } catch (err: any) {
    logger.error('Registration error', { error: err.message, correlationId: (req as any).correlationId });
    return res.status(400).json({ success: false, message: err.message });
  }
});

app.post('/api/v1/auth/login', async (req: Request, res: Response) => {
  try {
    const parsed = LoginSchema.parse(req.body);
    const user = await prisma.user.findUnique({ where: { email: parsed.email } });
    if (!user) {
      return res.status(401).json({ success: false, message: 'Invalid credentials' });
    }

    const passwordMatch = await verifyPassword(parsed.password, user.passwordHash);
    if (!passwordMatch) {
      return res.status(401).json({ success: false, message: 'Invalid credentials' });
    }

    const { accessToken, refreshToken } = generateTokens(user);
    await redis.setex(`refresh_${user.id}`, 7 * 24 * 60 * 60, refreshToken);

    logger.info('User logged in', { userId: user.id, correlationId: (req as any).correlationId });
    return res.json({
      success: true,
      user: { id: user.id, email: user.email, name: user.name },
      accessToken,
      refreshToken,
    });
  } catch (err: any) {
    logger.error('Login error', { error: err.message, correlationId: (req as any).correlationId });
    return res.status(400).json({ success: false, message: err.message });
  }
});

app.post('/api/v1/auth/refresh', async (req: Request, res: Response) => {
  try {
    const { refreshToken } = req.body;
    if (!refreshToken) {
      return res.status(401).json({ success: false, message: 'No refresh token' });
    }

    const decoded = jwt.verify(refreshToken, process.env.JWT_REFRESH_SECRET || 'dev-refresh-secret') as JWTPayload;
    const storedToken = await redis.get(`refresh_${decoded.userId}`);
    if (storedToken !== refreshToken) {
      return res.status(401).json({ success: false, message: 'Invalid refresh token' });
    }

    const user = await prisma.user.findUnique({ where: { id: decoded.userId } });
    if (!user) {
      return res.status(401).json({ success: false, message: 'User not found' });
    }

    const tokens = generateTokens(user);
    await redis.setex(`refresh_${user.id}`, 7 * 24 * 60 * 60, tokens.refreshToken);

    return res.json({ success: true, accessToken: tokens.accessToken, refreshToken: tokens.refreshToken });
  } catch (err: any) {
    logger.error('Refresh error', { error: err.message });
    return res.status(401).json({ success: false, message: 'Invalid refresh token' });
  }
});

app.post('/api/v1/payments/checkout', authMiddleware, async (req: Request, res: Response) => {
  try {
    const userId = (req as any).user.userId;
    const { plan } = req.body;

    // Create Razorpay payment link
    const paymentLinkRes = await axios.post(
      'https://api.razorpay.com/v1/payment_links',
      {
        upi_link: false,
        amount: plan === 'pro' ? 29 * 100 : plan === 'team' ? 149 * 100 : 0,
        currency: 'INR',
        accept_partial: false,
        description: `CodeConclave ${plan.toUpperCase()} Plan`,
        customer_notify: 1,
        notify: { sms: false, email: true },
        reminder_enable: true,
        notes: {
          userId,
          plan,
          orgId: (req as any).user.orgId,
        },
      },
      {
        auth: {
          username: process.env.RAZORPAY_KEY_ID || '',
          password: process.env.RAZORPAY_KEY_SECRET || '',
        },
      }
    );

    const paymentLink = paymentLinkRes.data;

    // Store in DB for verification
    await prisma.payment.create({
      data: {
        id: paymentLink.id,
        userId,
        plan: plan as PlanType,
        amount: paymentLink.amount,
        status: 'INITIATED',
        externalId: paymentLink.id,
        externalData: paymentLink,
      },
    });

    // Queue verification job
    await paymentQueue.add('verify-payment', {
      paymentLinkId: paymentLink.id,
      userId,
      plan,
    }, {
      delay: 5000, // Poll after 5s
      repeat: { every: 30000, limit: 120 }, // Poll every 30s for 1 hour
    });

    logger.info('Payment link created', { paymentLinkId: paymentLink.id, userId, plan });
    return res.json({
      success: true,
      shortUrl: paymentLink.short_url,
      paymentLinkId: paymentLink.id,
    });
  } catch (err: any) {
    logger.error('Payment checkout error', { error: err.message });
    return res.status(400).json({ success: false, message: err.message });
  }
});

app.post('/api/v1/payments/verify', authMiddleware, async (req: Request, res: Response) => {
  try {
    const userId = (req as any).user.userId;
    const { paymentLinkId } = req.body;

    const payment = await prisma.payment.findUnique({ where: { id: paymentLinkId } });
    if (!payment) {
      return res.status(404).json({ success: false, message: 'Payment not found' });
    }

    // Check payment status with Razorpay
    const statusRes = await axios.get(
      `https://api.razorpay.com/v1/payment_links/${paymentLinkId}`,
      {
        auth: {
          username: process.env.RAZORPAY_KEY_ID || '',
          password: process.env.RAZORPAY_KEY_SECRET || '',
        },
      }
    );

    if (statusRes.data.status === 'issued') {
      return res.json({ success: true, status: 'pending', message: 'Payment not yet completed' });
    }

    if (statusRes.data.status === 'paid') {
      // Update payment and entitlement
      await prisma.payment.update({
        where: { id: paymentLinkId },
        data: { status: 'COMPLETED' },
      });

      const expiresAt = new Date();
      if (payment.plan === 'pro') expiresAt.setMonth(expiresAt.getMonth() + 1);
      else if (payment.plan === 'team') expiresAt.setMonth(expiresAt.getMonth() + 1);

      await prisma.entitlement.update({
        where: { userId },
        data: {
          plan: payment.plan,
          tokensPerDay: payment.plan === 'pro' ? 50000 : payment.plan === 'team' ? 500000 : 500,
          interactionsPerDay: payment.plan === 'free' ? 2 : -1,
          projectsLimit: payment.plan === 'pro' ? 10 : payment.plan === 'team' ? -1 : 1,
          asyncHoursPerMonth: payment.plan === 'pro' ? 8 : payment.plan === 'team' ? 40 : 0,
          expiresAt,
        },
      });

      logger.info('Payment completed and entitlement updated', { userId, plan: payment.plan });
      return res.json({
        success: true,
        status: 'completed',
        entitlement: { plan: payment.plan },
      });
    }

    return res.json({ success: true, status: 'unknown' });
  } catch (err: any) {
    logger.error('Payment verification error', { error: err.message });
    return res.status(400).json({ success: false, message: err.message });
  }
});

// ==================== ENTITLEMENT ROUTES ====================
app.get('/api/v1/payments/entitlement', authMiddleware, async (req: Request, res: Response) => {
  try {
    const userId = (req as any).user.userId;

    const entitlement = await prisma.entitlement.findUnique({ where: { userId } });
    if (!entitlement) {
      return res.status(404).json({ success: false, message: 'Entitlement not found' });
    }

    // Get usage from Redis
    const tokensUsedKey = `tokens_used:${userId}:${new Date().toISOString().split('T')[0]}`;
    const tokensUsed = parseInt(await redis.get(tokensUsedKey) || '0', 10);
    const tokensRemaining = Math.max(0, entitlement.tokensPerDay - tokensUsed);

    const interactionsUsedKey = `interactions_used:${userId}:${new Date().toISOString().split('T')[0]}`;
    const interactionsUsed = parseInt(await redis.get(interactionsUsedKey) || '0', 10);
    const interactionsRemaining = entitlement.interactionsPerDay === -1 ? -1 : Math.max(0, entitlement.interactionsPerDay - interactionsUsed);

    return res.json({
      success: true,
      entitlement,
      limits: {
        tokensRemaining,
        interactionsRemaining,
      },
    });
  } catch (err: any) {
    logger.error('Entitlement fetch error', { error: err.message });
    return res.status(400).json({ success: false, message: err.message });
  }
});

// ==================== WORKER SETUP ====================
const taskWorker = new Worker('tasks', async (job) => {
  try {
    logger.info('Processing task', { jobId: job.id, taskId: job.data.taskId });

    // Simulate AI orchestration
    const task = await prisma.task.update({
      where: { id: job.data.taskId },
      data: { status: 'PLANNING' },
    });

    // Route to appropriate AI model based on task type
    let coworker = 'Architect';
    if (job.data.title.toLowerCase().includes('code')) coworker = 'Coder';
    else if (job.data.title.toLowerCase().includes('test')) coworker = 'Tester';

    await prisma.task.update({
      where: { id: job.data.taskId },
      data: { status: 'RUNNING', assignedCoworker: coworker },
    });

    // Simulate work
    await new Promise(resolve => setTimeout(resolve, 5000));

    await prisma.task.update({
      where: { id: job.data.taskId },
      data: {
        status: 'VERIFYING',
        result: { output: 'Task completed by ' + coworker },
      },
    });

    logger.info('Task completed', { taskId: job.data.taskId });
  } catch (err) {
    logger.error('Task processing error', { error: (err as any).message });
    throw err;
  }
}, { connection: redis });

const paymentWorker = new Worker('payment-verification', async (job) => {
  try {
    const { paymentLinkId, userId, plan } = job.data;

    const statusRes = await axios.get(
      `https://api.razorpay.com/v1/payment_links/${paymentLinkId}`,
      {
        auth: {
          username: process.env.RAZORPAY_KEY_ID || '',
          password: process.env.RAZORPAY_KEY_SECRET || '',
        },
      }
    );

    if (statusRes.data.status === 'paid') {
      await prisma.payment.update({
        where: { id: paymentLinkId },
        data: { status: 'COMPLETED' },
      });

      await prisma.entitlement.update({
        where: { userId },
        data: { plan: plan as PlanType },
      });

      logger.info('Auto-payment verified', { paymentLinkId });
      return { verified: true };
    }
  } catch (err) {
    logger.error('Payment verification job error', { error: (err as any).message });
  }
}, { connection: redis });

// Graceful shutdown
const gracefulShutdown = async (signal: string) => {
  logger.info(`Received ${signal}, shutting down gracefully`);
  server.close(async () => {
    await prisma.$disconnect();
    await redis.disconnect();
    await neo4j_driver.close();
    process.exit(0);
  });
};

process.on('SIGTERM', () => gracefulShutdown('SIGTERM'));
process.on('SIGINT', () => gracefulShutdown('SIGINT'));
