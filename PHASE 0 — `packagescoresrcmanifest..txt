export type SqlType = "uuid" | "text" | "citext" | "timestamptz" | "numeric" | "integer" | "boolean" | "jsonb";

export interface ColumnContract {

  readonly type: SqlType;
  readonly nullability: Nullability;
  readonly default?: string;
  readonly primaryKey?: boolean;
  readonly references?: { readonly table: string; readonly column: string; readonly onDelete: "CASCADE" | "SET NULL" | "RESTRICT" };
  readonly unique?: boolean;
  readonly check?: string;
  readonly index?: boolean;
}

export interface RlsPolicy {
  readonly forced: boolean;
  readonly using: string;
  readonly withCheck: string;
}

export interface TableContract {
  readonly table: string;
  readonly purpose: string;
  readonly columns: readonly ColumnContract[];
  readonly primaryKey: readonly string[];
  readonly foreignKeys: readonly { readonly column: string; readonly references: { readonly table: string; readonly column: string; readonly onDelete: "CASCADE" | "SET NULL" | "RESTRICT" } }[];
  readonly uniqueConstraints: readonly { readonly name: string; readonly columns: readonly string[] }[];
  readonly checkConstraints: readonly { readonly name: string; readonly expression: string }[];
  readonly indexes: readonly { readonly name: string; readonly columns: readonly string[]; readonly unique?: boolean }[];
  readonly rls: RlsPolicy | null;
  readonly stateMachine: string | null;
  readonly retention: string | null;
  readonly audit: "APPEND_ONLY" | "TRACE" | null;
}

class TableContractBuilder {
  private readonly c: TableContract;

  constructor(table: string, purpose: string) {
    this.c = { table, purpose, columns: [], primaryKey: [], foreignKeys: [], uniqueConstraints: [], checkConstraints: [], indexes: [], rls: null, stateMachine: null, retention: null, audit: null };
  }

  column(name: string, type: SqlType, nullability: Nullability = "NOT NULL"): this {
    this.c.columns = [...this.c.columns, { name, type, nullability }];
    return this;
  }

  withDefault(expr: string): this {
    const last = this.c.columns[this.c.columns.length - 1];
    this.c.columns = [...this.c.columns.slice(0, -1), { ...last, default: expr }];
    return this;
  }

  primaryKey(...columns: string[]): this {
    this.c.primaryKey = columns;
    this.c.columns = this.c.columns.map((col) => (columns.includes(col.name) ? { ...col, primaryKey: true } : col));
    return this;
  }

  references(table: string, column: string, onDelete: "CASCADE" | "SET NULL" | "RESTRICT" = "RESTRICT"): this {
    const last = this.c.columns[this.c.columns.length - 1];
    const ref = { table, column, onDelete };
    this.c.columns = [...this.c.columns.slice(0, -1), { ...last, references: ref }];
    this.c.foreignKeys = [...this.c.foreignKeys, { column: last.name, references: ref }];
    return this;
  }

  unique(name: string, ...columns: string[]): this {
    this.c.uniqueConstraints = [...this.c.uniqueConstraints, { name, columns }];
    return this;
  }

  check(name: string, expression: string): this {
    this.c.checkConstraints = [...this.c.checkConstraints, { name, expression }];
    return this;
  }

  index(name: string, ...columns: string[]): this {
    this.c.indexes = [...this.c.indexes, { name, columns }];
    for (const col of columns) {
      this.c.columns = this.c.columns.map((x) => (x.name === col ? { ...x, index: true } : x));
    }
    return this;
  }

  rls(using: string, withCheck: string, forced = true): this {
    this.c.rls = { forced, using, withCheck };
    return this;
  }

  stateMachine(definition: string): this {
    this.c.stateMachine = definition;
    return this;
  }

  retention(definition: string): this {
    this.c.retention = definition;
    return this;
  }

  audit(mode: "APPEND_ONLY" | "TRACE"): this {
    this.c.audit = mode;
    return this;
  }

  build(): Readonly<TableContract> {
    return Object.freeze({ ...this.c, columns: Object.freeze([...this.c.columns]) });
  }
}

export function table(name: string, purpose: string): TableContractBuilder {
  return new TableContractBuilder(name, purpose);
}

import { createHmac, createCipheriv, createDecipheriv, randomBytes, scryptSync, timingSafeEqual } from "node:crypto";

const B32 = "ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";

export function base32Encode(bytes: Uint8Array): string {
  let bits = 0, value = 0, out = "";
  for (const b of bytes) {
    value = (value << 8) | b;
    bits += 8;
    while (bits >= 5) {
      out += B32[(value >>> (bits - 5)) & 31];
      bits -= 5;
    }
  }
  if (bits > 0) out += B32[(value << (5 - bits)) & 31];
  return out;
}

export function totpSecret(): string {
  return base32Encode(randomBytes(20));
}

export function totpCode(secretB32: string, at = Date.now()): string {
  const key = Buffer.from(secretB32.replace(/=+$/, ""), "base64url");
  const counter = Math.floor(at / 1000 / 30);
  const buf = Buffer.alloc(8);
  buf.writeBigUInt64BE(BigInt(counter));
  const h = createHmac("sha1", key).update(buf).digest();
  const o = h[h.length - 1] & 0x0f;
  return (((h[o] & 0x7f) << 24) | (h[o + 1] << 16) | (h[o + 2] << 8) | h[o + 3]) % 1_000_000
    .toString().padStart(6, "0");
}

export function verifyTOTP(secretB32: string, code: string, window = 1): boolean {
  for (let w = -window; w <= window; w++) {
    if (timingSafeEqual(Buffer.from(totpCode(secretB32, Date.now() + w * 30_000)), Buffer.from(code))) return true;
  }
  return false;
}

  readonly kind: MemoryKind;

  readonly embedding: readonly number[] | null;
  readonly vectorIndex: "pgvector_cosine" | null;
  readonly fts: string | null;

export function embed(content: string, dim: typeof MEMORY_DIM = MEMORY_DIM): readonly number[] {
  if (dim !== MEMORY_DIM) throw new Error("MEMORY LOCK: embeddings are 1536-dimensional.");
  return Array.from({ length: dim }, (_, i) => hashToUnit(content, i));
}

function hashToUnit(seed: string, i: number): number {
  let h = 2166136261 ^ seed.charCodeAt(i % seed.length);
  for (const c of seed) h = ((h ^ c.charCodeAt(0)) * 16777619) >>> 0;
  const v = Math.sin(h + i) * 10000;
  return v - Math.floor(v) * 2 - 1;
}

export function cosineSimilarity(a: readonly number[], b: readonly number[]): number {
  if (a.length !== b.length || a.length !== MEMORY_DIM) throw new Error("MEMORY LOCK: dimension mismatch.");
  let dot = 0, na = 0, nb = 0;
  for (let i = 0; i < a.length; i++) {
    dot += a[i] * b[i];
    na += a[i] * a[i];
    nb += b[i] * b[i];
  }
  return dot / (Math.sqrt(na) * Math.sqrt(nb) || 1);
}

export interface DnaSnapshot {
  readonly projectId: string;
  readonly version: number;
  readonly baseVersion: number | null;
  readonly filesReferenced: readonly string[];
  readonly decisions: ReadonlyArray<{ readonly key: string; readonly value: string }>;
  readonly unresolvedTasks: readonly string[];
  readonly nextActions: readonly string[];
  readonly environmentState: Readonly<Record<string, string>>;
  readonly testResults: ReadonlyArray<{ readonly suite: string; readonly passed: boolean }>;
}

export function nextDnaVersion(snapshots: readonly DnaSnapshot[]): number {
  return snapshots.reduce((max, s) => Math.max(max, s.version), 0) + 1;
}

export type DnaConflict =
  | { readonly key: string; readonly base: string; readonly ours: string; readonly theirs: string }
  | null;

export function detectConflict(base: DnaSnapshot, ours: DnaSnapshot, theirs: DnaSnapshot): readonly DnaConflict[] {
  if (ours.baseVersion !== theirs.baseVersion) throw new Error("DNA LOCK: divergent branches must share a base version.");
  const conflicts: DnaConflict[] = [];
  const all = new Set([...Object.keys(base.decisions), ...Object.keys(ours.decisions), ...Object.keys(theirs.decisions)]);
  for (const key of all) {
    const b = base.decisions.find((d) => d.key === key)?.value;
    const o = ours.decisions.find((d) => d.key === key)?.value;
    const t = theirs.decisions.find((d) => d.key === key)?.value;
    if (o !== undefined && t !== undefined && o !== t && o !== b && t !== b) {
      conflicts.push({ key, base: b ?? "", ours: o, theirs: t });
    }
  }
  return conflicts;
}

export type MergeResolution = "OURS" | "THEIRS" | "MANUAL" | "MOST_RECENT_WINS";

export function resolveConflict(conflict: DnaConflict, resolution: MergeResolution, manualValue?: string): string {
  switch (resolution) {
    case "OURS": return conflict.ours;
    case "THEIRS": return conflict.theirs;
    case "MOST_RECENT_WINS": return conflict.ours; // timestamp tiebreak applied by caller
    case "MANUAL":
      if (manualValue === undefined) throw new Error("DNA LOCK: manual merge requires a value.");
      return manualValue;
  }
}

export function mergedSnapshot(base: DnaSnapshot, ours: DnaSnapshot, theirs: DnaSnapshot, resolutions: Readonly<Record<string, MergeResolution>>, manual: Readonly<Record<string, string>> = {}): DnaSnapshot {
  const conflicts = detectConflict(base, ours, theirs);
  const mergedDecisions = [...theirs.decisions];
  for (const c of conflicts) {
    const value = resolveConflict(c, resolutions[c.key] ?? "MANUAL", manual[c.key]);
    mergedDecisions = [...mergedDecisions.filter((d) => d.key !== c.key), { key: c.key, value }];
  }
  return {
    projectId: theirs.projectId,
    version: nextDnaVersion([base, ours, theirs]),
    baseVersion: theirs.baseVersion,
    filesReferenced: [...new Set([...ours.filesReferenced, ...theirs.filesReferenced])],
    decisions: mergedDecisions,
    unresolvedTasks: [...new Set([...ours.unresolvedTasks, ...theirs.unresolvedTasks])],
    nextActions: [...new Set([...ours.nextActions, ...theirs.nextActions])],
    environmentState: { ...ours.environmentState, ...theirs.environmentState },
    testResults: [...ours.testResults, ...theirs.testResults],
  };
}

export type IntentPath = "FAST" | "DEEP";

export interface IntentInput {
  readonly message: string;
  readonly attachedFileCount: number;
  readonly currentMode: "CHAT" | "COWORK";

export interface IntentDecision {
  readonly path: IntentPath;
  readonly confidence: number;
  readonly triggers: readonly string[];

const FAST_TRIGGERS: readonly { readonly pattern: RegExp; readonly label: string; readonly weight: number }[] = [
  { pattern: /^what is\b/i, label: "simple question", weight: 1 },
  { pattern: /\bexplain\b/i, label: "explanation", weight: 1 },
  { pattern: /\bsummariz\w+\b/i, label: "summarization", weight: 1 },
  { pattern: /\bfix this file\b/i, label: "single file fix", weight: 1 },
  { pattern: /\bread (the )?(first|few|this)\b/i, label: "read files", weight: 1 },
  { pattern: /\bquick (lookup|look)\b/i, label: "quick lookup", weight: 1 },
  { pattern: /\blightweight\b/i, label: "lightweight command", weight: 1 },
];

const DEEP_TRIGGERS: readonly { readonly pattern: RegExp; readonly label: string; readonly weight: number }[] = [
  { pattern: /\brefactor\b/i, label: "refactor", weight: 3 },
  { pattern: /\barchitecture\b/i, label: "architecture", weight: 3 },
  { pattern: /design (the )?(architecture|system) (and|then) (implement|build)/i, label: "design and implement", weight: 4 },
  { pattern: /\bsecurity review\b/i, label: "security review", weight: 3 },
  { pattern: /\b(multi-file|multiple files|all files)\b/i, label: "multi-file task", weight: 4 },
  { pattern: /\b(plan|planning|long-running|long running|autonomous)\b/i, label: "planning or autonomy", weight: 2 },
  { pattern: /\b(implement|build|create)\b.*\b(module|system|feature|service)\b/i, label: "implementation", weight: 3 },
  { pattern: /\brun (all|the) tests\b/i, label: "testing", weight: 2 },
];

export function classifyIntent(input: IntentInput): IntentDecision {
  if (input.currentMode === "COWORK") {
    return { path: "DEEP", confidence: 1, triggers: ["cowork mode"] };
  }
  if (input.attachedFileCount > 1) {
    return { path: "DEEP", confidence: 1, triggers: ["multiple attachments"] };
  }
  const fast = FAST_TRIGGERS.map((t) => ({ t, m: input.message.match(t.pattern) })).filter((x) => x.m !== null);
  const deep = DEEP_TRIGGERS.map((t) => ({ t, m: input.message.match(t.pattern) })).filter((x) => x.m !== null);
  const fastScore = fast.reduce((s, x) => s + x.t.weight, 0);
  const deepScore = deep.reduce((s, x) => s + x.t.weight, 0);
  if (deepScore > fastScore) {
    return { path: "DEEP", confidence: deepScore / (deepScore + fastScore || 1), triggers: deep.map((x) => x.t.label) };
  }
  if (fastScore > 0) {
    return { path: "FAST", confidence: fastScore / (deepScore + fastScore || 1), triggers: fast.map((x) => x.t.label) };
  }
  return { path: "FAST", confidence: 0, triggers: ["default fast path"] };
}

export type ParallelPair = readonly [CoworkerId, CoworkerId];

export const PARALLEL_SAFE_PAIRS: readonly Readonly<{ readonly pair: ParallelPair; readonly requires: "NONE" | "IMPLEMENTATION_COMPLETE" | "STABLE_INPUTS" }>[] = [
  { pair: ["ARCHITECT", "RESEARCH"], requires: "NONE" },
  { pair: ["TESTER", "PERFORMANCE"], requires: "IMPLEMENTATION_COMPLETE" },
  { pair: ["SECURITY", "REVIEWER"], requires: "STABLE_INPUTS" },
];

export interface WorkerPlan {
  readonly id: string;
  readonly coworker: CoworkerId;
  readonly writesTo: readonly string[];
}

export function canCorruptState(plan: readonly WorkerPlan[]): boolean {
  const targets = new Map<string, string>();
  for (const worker of plan) {
    for (const target of worker.writesTo) {
      if (targets.has(target) && targets.get(target) !== worker.id) return true;
      targets.set(target, worker.id);
    }
  }
  return false;
}

export function assertParallelSafe(a: CoworkerId, b: CoworkerId, implementationComplete: boolean, inputsStable: boolean): void {
  const rule = PARALLEL_SAFE_PAIRS.find((r) => r.pair.includes(a) && r.pair.includes(b));
  if (!rule) throw new Error(`WORKFLOW LOCK: ${a} + ${b} must never run in parallel.`);
  if (rule.requires === "IMPLEMENTATION_COMPLETE" && !implementationComplete) {
    throw new Error(`WORKFLOW LOCK: ${a} + ${b} may run in parallel only after implementation completes.`);
  }
  if (rule.requires === "STABLE_INPUTS" && !inputsStable) {
    throw new Error(`WORKFLOW LOCK: ${a} + ${b} may run in parallel only when inputs are stable.`);
  }
}

export function assertNoStateCorruption(plan: readonly WorkerPlan[]): void {
  if (canCorruptState(plan)) {
    throw new Error("WORKFLOW LOCK: never parallelize operations that can corrupt state.");
  }
}

export function Composer(props: ComposerProps): JSX.Element {
  const { mode, onModeChange, model, plan, onModelSelect, onSend, onCancelSuggestion, onVoiceInput } = props;
  const placeholder = props.placeholder ?? "Ask CodeConClave anything or give a command...";
  const [text, setText] = useState("");
  const [attachments, setAttachments] = useState<readonly ChatAttachment[]>([]);
  const fileInput = useRef<HTMLInputElement>(null);

  const addFiles = (files: readonly File[]): void => {
    setAttachments((prev) => [
      ...prev,
      ...files.map((f) => ({
        id: crypto.randomUUID(),
        name: f.name,
        kind: f.type.startsWith("image/") ? ("image" as const) : f.name.endsWith(".ts") || f.name.endsWith(".js") ? ("code" as const) : ("file" as const),

import type { TerminalTab } from "./terminal";

export type TerminalShell = "bash" | "zsh" | "node" | "python" | "powershell";

export type ProcessStatusKind = "RUNNING" | "STOPPED" | "COMPLETED" | "FAILED";

export function statusKind(tab: TerminalTab, live: boolean): ProcessStatusKind {
  if (tab.status === "killed") return "STOPPED";
  if (tab.status === "idle") return "COMPLETED";
  if (tab.exitCode !== null && tab.exitCode !== 0) return "FAILED";
  return live ? "RUNNING" : "COMPLETED";
}

export const STATUS_GLYPH: Readonly<Record<ProcessStatusKind, string>> = {
  RUNNING: "●",
  STOPPED: "○",
  COMPLETED: "✓",
  FAILED: "✗",

export interface TerminalPanelState {
  readonly tabs: readonly TerminalTab[];
  readonly activeTabId: string;
  readonly history: readonly string[];
  readonly historyIndex: number;
}

export function createPanel(shell: TerminalShell = "bash", cwd = process.cwd()): TerminalPanelState {
  return {
    tabs: [startProcess(shell, cwd)],
    activeTabId: "",
    history: [],
    historyIndex: -1,
  };
}

export function pushHistory(state: TerminalPanelState, command: string): TerminalPanelState {
  return { ...state, history: [...state.history, command], historyIndex: -1 };
}

export function historyPrev(state: TerminalPanelState): string {
  const idx = Math.min(state.historyIndex + 1, state.history.length - 1);
  return idx >= 0 ? state.history[idx] : "";
}

export function historyNext(state: TerminalPanelState): string {
  const idx = Math.max(state.historyIndex - 1, -1);
  return idx >= 0 ? state.history[idx] : "";
}

export interface TreeNode {
  readonly path: string;
  readonly name: string;
  readonly kind: "directory" | "file";
  readonly children: readonly TreeNode[];
  readonly meta: FileMeta | null;
}

export function buildTree(paths: readonly { readonly path: string; readonly meta: FileMeta | null }[], root = "~"): TreeNode {
  const rootNode: TreeNode = { path: root, name: root, kind: "directory", children: [], meta: null };
  const dirs = new Map<string, TreeNode>();
  dirs.set(root, rootNode);
  for (const p of paths) {
    const parts = p.path.split("/").filter((s) => s.length > 0);
    let current = rootNode;
    let acc = root;
    for (let i = 0; i < parts.length; i++) {
      acc += "/" + parts[i];
      const isFile = i === parts.length - 1 && p.meta !== null;
      let node = dirs.get(acc);
      if (!node) {
        node = { path: acc, name: parts[i], kind: isFile ? "file" : "directory", children: [], meta: isFile ? p.meta : null };
        dirs.set(acc, node);
        current.children = [...current.children, node];
      }
      current = node;
    }
  }
  return rootNode;
}

export function sortTree(node: TreeNode): TreeNode {
  const children = [...node.children].map(sortTree).sort((a, b) => {
    if (a.kind !== b.kind) return a.kind === "directory" ? -1 : 1;
    return a.name.localeCompare(b.name);
  });
  return { ...node, children };
}

export function unifiedDiff(beforeLines: readonly string[], afterLines: readonly string[], context = 3): string {
  const n = beforeLines.length, m = afterLines.length;
  const dp: number[][] = Array.from({ length: n + 1 }, () => Array.from({ length: m + 1 }, () => 0));
  for (let i = n - 1; i >= 0; i--) {
    for (let j = m - 1; j >= 0; j--) {
      dp[i][j] = beforeLines[i] === afterLines[j] ? dp[i + 1][j + 1] + 1 : Math.max(dp[i + 1][j], dp[i][j + 1]);
    }
  }
  const out: string[] = [];
  let i = 0, j = 0;
  while (i < n && j < m) {
    if (beforeLines[i] === afterLines[j]) {
      out.push(" " + beforeLines[i]);
      i++; j++;
    } else if (dp[i + 1][j] >= dp[i][j + 1]) {
      out.push("-" + beforeLines[i]);
      i++;
    } else {
      out.push("+" + afterLines[j]);
      j++;
    }
  }
  while (i < n) { out.push("-" + beforeLines[i]); i++; }
  while (j < m) { out.push("+" + afterLines[j]); j++; }
  return out.join("\n");
}

export const PIPELINE_DEFAULT_TIMEOUT_MS = 10 * 60 * 1000;

export interface PipelineStep {
  readonly id: string;
  readonly coworker: CoworkerId;
  readonly requirements: string;
  readonly dependencies: readonly string[];
  readonly approvalGate: boolean;
  readonly timeoutMs: number;
  readonly maxTokens: number;
  readonly acceptanceCriteria: string;
}

export interface Pipeline {
  readonly id: string;
  readonly taskId: string;
  readonly steps: readonly PipelineStep[];
}

export const CANONICAL_PIPELINE: readonly CoworkerId[] = ["ARCHITECT", "CODER", "TESTER", "SECURITY", "REVIEWER"];

export function topologicalOrder(pipeline: Pipeline): readonly PipelineStep[] {
  const byId = new Map(pipeline.steps.map((s) => [s.id, s]));
  const visiting = new Set<string>();
  const visited = new Set<string>();
  const out: PipelineStep[] = [];
  const visit = (step: PipelineStep): void => {
    if (visited.has(step.id)) return;
    if (visiting.has(step.id)) throw new Error(`PIPELINE LOCK: dependency cycle at step ${step.id}.`);
    visiting.add(step.id);
    for (const dep of step.dependencies) {
      const depStep = byId.get(dep);
      if (depStep) visit(depStep);
    }
    visiting.delete(step.id);
    visited.add(step.id);
    out.push(step);
  };
  for (const step of pipeline.steps) visit(step);
  return out;
}

export function parallelizable(step: PipelineStep, order: readonly PipelineStep[]): boolean {
  return step.dependencies.length === 0;
}

export interface HandoffRecord {
  readonly taskId: string;
  readonly requirements: string;
  readonly context: string;
  readonly files: readonly string[];
  readonly artifacts: readonly string[];
  readonly previousResult: string | null;
  readonly acceptanceCriteria: string;
  readonly limitations: readonly string[];
  readonly verificationStatus: "PENDING" | "VERIFIED" | "FAILED";
}

export function createHandoff(step: PipelineStep, taskId: string): HandoffRecord {
  return {
    taskId,
    requirements: step.requirements,
    context: "",
    files: [],
    artifacts: [],
    previousResult: null,
    acceptanceCriteria: step.acceptanceCriteria,
    limitations: [],
    verificationStatus: "PENDING",
  };
}

export type FailureRecovery = "SKIP_WITH_FALLBACK" | "PAUSE_PIPELINE";

export interface StepFailure {
  readonly stepId: string;
  readonly coworker: CoworkerId;
  readonly reason: string;
  readonly recovery: FailureRecovery;
}

export function recoverFromStepFailure(failure: StepFailure, pipeline: Pipeline, fallback: CoworkerId | null): readonly PipelineStep[] {
  if (failure.recovery === "PAUSE_PIPELINE") throw new Error(`PIPELINE LOCK: pipeline paused at step ${failure.stepId}.`);
  if (!fallback) throw new Error("PIPELINE LOCK: skip-with-fallback requires a fallback coworker.");
  return pipeline.steps.map((s) => (s.id === failure.stepId ? { ...s, coworker: fallback } : s));
}

export function qualifiedCheaper(requested: SelectableModelId, klass: ComputeClass): SelectableModelId | null {
  const preferred = COMPUTE_CLASS_RULES[klass].preferred;
  if (preferred.includes(requested)) return null;
  return preferred[0] ?? null;

export function computeDecision(ctx: ComputeContext): ComputeDecision {
  const isPremium = MODEL_GROUPS[0].label === "Premium" && ctx.requestedClass === "C_PREMIUM";
  const premiumExhausted = ctx.budget.maxPremiumCallsPerPeriod >= 0 && ctx.premiumCallsUsedThisPeriod >= ctx.budget.maxPremiumCallsPerPeriod;
  if (isPremium && premiumExhausted) {
    const cheaper = qualifiedCheaper(ctx.requestedModel, "B_STANDARD");
    if (cheaper && ctx.budget.fallbackPolicy === "ALWAYS_FALLBACK_WHEN_QUALIFIED") {
      return { action: "FALLBACK", model: cheaper, reason: "premium quota exhausted; routed to qualified cheaper model" };
    }
    if (cheaper) {
      return { action: "EXTENDED_COMPUTE", reason: "premium compute quota exhausted for this period", estimatedCostUsd: estimateCostUsd(cheaper, 12_000) };
    }
    return { action: "BLOCK", reason: "no qualified cheaper model available; extended compute required" };
  }
  return { action: "EXECUTE", model: ctx.requestedModel, reason: "entitlement and compute policy satisfied" };
}

export const APPROVAL_TRANSITIONS: Readonly<Record<ApprovalStatus, readonly ApprovalStatus[]>> = {

  REJECTED: ["CANCELLED"],
  CANCELLED: [],
  EXPIRED: [],
};

export function transitionApproval(card: ApprovalCard, to: ApprovalStatus): ApprovalCard {
  if (!APPROVAL_TRANSITIONS[card.status].includes(to)) {
    throw new Error(`APPROVAL LOCK: illegal transition ${card.status} -> ${to}`);
  }
  return { ...card, status: to };
}

export function maybeExpire(card: ApprovalCard, now = Date.now()): ApprovalCard {
  if (card.status === "PENDING" && now > card.expiresAt) {
    return { ...card, status: "EXPIRED" };
  }
  return card;
}

export interface ModelRegistryRow {
  readonly modelId: string;
  readonly provider: string;
  readonly displayName: string;
  readonly tier: "PREMIUM" | "CAPABLE" | "EFFICIENT";
  readonly contextWindow: number;
  readonly visionSupported: boolean;
  readonly toolSupport: boolean;
  readonly functionCalling: boolean;
  readonly inputPricePer1K: number;
  readonly outputPricePer1K: number;
  readonly planEligibility: readonly string[];
  readonly privacyClass: string;
  readonly targetLatencyMs: number;
  readonly healthStatus: "healthy" | "degraded" | "down" | "unknown";
  readonly priority: number;
  readonly fallbackModels: readonly string[];
  readonly enabled: boolean;
  readonly effectiveDate: number;
  readonly deprecatedDate: number | null;
}

export interface RoutingRequirements {
  readonly plan: string;
  readonly vision: boolean;
  readonly tools: boolean;
  readonly minContextWindow: number;
  readonly latencySensitive: boolean;
}

export function routeModels(registry: readonly ModelRegistryRow[], req: RoutingRequirements): readonly ModelRegistryRow[] {
  const eligible = registry.filter((m) => m.enabled && m.planEligibility.includes(req.plan) && (m.deprecatedDate === null || m.deprecatedDate > Date.now()));
  const capable = eligible.filter((m) => (!req.vision || m.visionSupported) && (!req.tools || m.toolSupport) && m.contextWindow >= req.minContextWindow);
  const pool = capable.length > 0 ? capable : eligible;
  return [...pool].sort((a, b) => {
    if (req.latencySensitive && (a.healthStatus === "down" ? 1 : 0) !== (b.healthStatus === "down" ? 1 : 0)) return (a.healthStatus === "down" ? 1 : 0) - (b.healthStatus === "down" ? 1 : 0);
    if (req.latencySensitive && a.healthStatus !== b.healthStatus) return a.healthStatus === "down" ? 1 : b.healthStatus === "down" ? -1 : a.healthStatus === "degraded" ? 1 : -1;
    return (a.inputPricePer1K + a.outputPricePer1K) / 2 - (b.inputPricePer1K + b.outputPricePer1K) / 2 || b.priority - a.priority;
  });
}

export function top3(registry: readonly ModelRegistryRow[], req: RoutingRequirements): readonly [ModelRegistryRow, ModelRegistryRow, ModelRegistryRow] {
  const ranked = routeModels(registry, req);
  if (ranked.length < 3) throw new Error("MODEL GATEWAY: fewer than 3 eligible models in registry.");
  return [ranked[0], ranked[1], ranked[2]];
}

export class AIGateway {
  constructor(private readonly adapters: Readonly<Record<string, ProviderAdapter>>) {}

  async callWithFallback(chain: readonly ModelRegistryRow[], request: NormalizedRequest): Promise<NormalizedResponse> {
    const errors: string[] = [];
    for (const row of chain) {
      const adapter = this.adapters[row.provider];
      if (!adapter) { errors.push(`${row.provider}: no adapter`); continue; }
      if (row.healthStatus === "down") { errors.push(`${row.modelId}: health down`); continue; }
      try {
        return await adapter.call({ model: row.modelId, ...request });
      } catch (e) {
        errors.push(`${row.modelId}: ${String(e)}`);
      }
    }
    throw new Error(`MODEL GATEWAY: graceful failure — ${errors.join(" | ")}`);
  }
}

  readonly inputTokens: number;
  readonly outputTokens: number;
}

  readonly call: (req: NormalizedRequest & { readonly model: string }) => Promise<NormalizedResponse>;
  readonly health: () => Promise<boolean>;
}

export interface AuditBlock {
  readonly index: number;
  readonly prevHash: string;
  readonly payload: string;
  readonly hash: string;
}

export function chainAudit(blocks: readonly AuditBlock[], payload: string): AuditBlock {
  const last = blocks[blocks.length - 1];
  const index = (last?.index ?? -1) + 1;
  const prevHash = last?.hash ?? createHash("sha256").update("GENESIS").digest("hex");
  const hash = createHash("sha256").update(`${index}|${prevHash}|${payload}`).digest("hex");
  return { index, prevHash, payload, hash };
}

export function verifyAuditChain(blocks: readonly AuditBlock[]): boolean {
  return blocks.every((b, i) => {
    const prev = i === 0 ? createHash("sha256").update("GENESIS").digest("hex") : blocks[i - 1].hash;
    return b.index === i && b.prevHash === prev && b.hash === createHash("sha256").update(`${b.index}|${b.prevHash}|${b.payload}`).digest("hex");
  });
}
